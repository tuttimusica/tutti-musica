-- REVIEW-ONLY DRAFT. DO NOT RUN ON PRODUCTION WITHOUT BACKUP, STAGING TESTS, AND EXPLICIT APPROVAL.
-- Issue: private.tutti_is_admin() checks authorized email, but private.is_admin()
-- checks usuarios.role='admin'. The approval RPC updates usuarios.role and invokes
-- private.protect_user_role(), which calls private.is_admin(). With zero admin
-- usuarios rows, a legitimate developer cannot approve teachers.
--
-- Proposed change: unify private.is_admin() with the authoritative allowlist,
-- bound to the authenticated user's UUID AND confirmed email in auth.users.
-- No client-supplied role, user metadata, or email is trusted.
-- Important: audit all callers of private.is_admin() before applying: this
-- function also controls existing RLS policies in other tables.
--
-- STAGING ONLY:
CREATE OR REPLACE FUNCTION private.is_admin()
RETURNS boolean
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path TO ''
AS $$
  SELECT EXISTS (
    SELECT 1
    FROM auth.users AS au
    JOIN public.administradores_autorizados AS aa
      ON lower(btrim(aa.email)) = lower(btrim(au.email))
    WHERE au.id = (SELECT auth.uid())
      AND au.email_confirmed_at IS NOT NULL
  );
$$;
